cmake_minimum_required(VERSION 3.24)
project(WizCommentFocusRegression LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core WebEngineCore)
set(source "${CMAKE_CURRENT_SOURCE_DIR}/../../src/WizMainWindow.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
file(READ "${source}" commands)
string(REPLACE "\r\n" "\n" commands "${commands}")
string(FIND "${commands}" "void WizMainWindow::on_actionEditingUndo_triggered()" begin)
string(FIND "${commands}" "void WizMainWindow::on_actionEditingDelete_triggered()" end)
if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "Review current edit-command extraction boundaries")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${commands}" ${begin} ${length} commands)
string(REPLACE "WizMainWindow::" "FocusHarness::" commands "${commands}")
string(REPLACE "WizDocumentView" "FakeDocument" commands "${commands}")
string(REPLACE "WizCodeEditorDialog" "FakeCodeEditor" commands "${commands}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/actual-commands.inc" "${commands}")
add_executable(comment-focus main.cpp)
target_include_directories(comment-focus PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(comment-focus PRIVATE cxx_std_17)
target_link_libraries(comment-focus PRIVATE Qt6::Core Qt6::WebEngineCore)
