cmake_minimum_required(VERSION 3.24)
project(WizCacheAdmissionPerformance LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
set(WIZ_PERF_CACHE_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/share" CACHE PATH "Header under test")
set(WIZ_PERF_EXPECTED_REDUNDANT 0 CACHE STRING "Expected admitted requests in the saved comparison header")
add_executable(cache-admission cache-admission.cpp)
target_include_directories(cache-admission PRIVATE "${WIZ_PERF_CACHE_INCLUDE}")
target_compile_definitions(cache-admission PRIVATE WIZ_PERF_EXPECTED_REDUNDANT=${WIZ_PERF_EXPECTED_REDUNDANT})
target_compile_features(cache-admission PRIVATE cxx_std_17)
target_link_libraries(cache-admission PRIVATE Qt6::Core Qt6::Gui)
