cmake_minimum_required(VERSION 3.24)
project(WizReaderOpeningDiagnostics LANGUAGES C CXX OBJCXX)
find_package(Qt6 REQUIRED COMPONENTS Widgets WebEngineWidgets WebChannel QuickWidgets LinguistTools)
set(CMAKE_AUTOMOC ON)
qt_wrap_ui(settings_ui ../../src/ui/WizPreferenceDialog.ui)
qt_add_translation(settings_qm ../../i18n/wiznote_zh_CN.ts)
add_executable(pdf-settings-labels settings.cpp ${settings_ui} ${settings_qm})
target_include_directories(pdf-settings-labels PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(pdf-settings-labels PRIVATE Qt6::Widgets)
target_compile_features(pdf-settings-labels PRIVATE cxx_std_17)
add_executable(pdf-export-menu ../compat/pdf-export-menu.cpp)
target_include_directories(pdf-export-menu PRIVATE ../../src)
target_link_libraries(pdf-export-menu PRIVATE Qt6::Widgets)
target_compile_features(pdf-export-menu PRIVATE cxx_std_17)
add_executable(reader-opening main.cpp ../compat/native-memory-pool.mm ../../src/WizNotePresentation.cpp)
target_link_libraries(reader-opening PRIVATE Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel Qt6::QuickWidgets "-framework Foundation")
target_include_directories(reader-opening PRIVATE ../../src)
target_compile_features(reader-opening PRIVATE cxx_std_17)
add_executable(category-selection category-selection.cpp)
target_include_directories(category-selection PRIVATE ../../src)
target_link_libraries(category-selection PRIVATE Qt6::Widgets)
target_compile_features(category-selection PRIVATE cxx_std_17)
add_executable(todo-storage todo-storage.cpp ../../src/database/sqlite3.c)
target_include_directories(todo-storage PRIVATE ../../src)
target_link_libraries(todo-storage PRIVATE Qt6::Widgets)
target_compile_features(todo-storage PRIVATE cxx_std_17)
target_compile_definitions(todo-storage PRIVATE SQLITE_THREADSAFE=1 SQLITE_DEFAULT_PAGE_SIZE=1024)
