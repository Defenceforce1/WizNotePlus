cmake_minimum_required(VERSION 3.24)
project(WizReaderOpeningDiagnostics LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Widgets WebEngineWidgets WebChannel QuickWidgets LinguistTools)
set(CMAKE_AUTOMOC ON)
qt_wrap_ui(settings_ui ../../src/ui/WizPreferenceDialog.ui)
qt_add_translation(settings_qm ../../i18n/wiznote_zh_CN.ts)
add_executable(pdf-settings-labels settings.cpp ${settings_ui} ${settings_qm})
target_include_directories(pdf-settings-labels PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(pdf-settings-labels PRIVATE Qt6::Widgets)
target_compile_features(pdf-settings-labels PRIVATE cxx_std_17)
add_executable(reader-opening main.cpp)
target_link_libraries(reader-opening PRIVATE Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel Qt6::QuickWidgets)
target_include_directories(reader-opening PRIVATE ../../src)
target_compile_features(reader-opening PRIVATE cxx_std_17)
