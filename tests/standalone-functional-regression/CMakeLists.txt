cmake_minimum_required(VERSION 3.24)
project(WizOfflineFunctionalRegression LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network WebEngineCore WebEngineWidgets Core5Compat)
set(CMAKE_AUTOMOC ON)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(compat "${root}/tests/compat")
function(wiz_unit name source)
  add_executable(${name} "${compat}/${source}")
  target_include_directories(${name} PRIVATE "${root}/src")
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_link_libraries(${name} PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Core5Compat)
endfunction()
wiz_unit(bounded-cache bounded-cache.cpp)
wiz_unit(thread-pool thread-pool.cpp)
wiz_unit(document-owner-actions document-owner-actions.cpp)
foreach(name thread-pool document-owner-actions)
  target_sources(${name} PRIVATE "${root}/src/share/WizThreads.cpp" "${root}/src/share/WizThreads_p.h")
endforeach()
target_compile_definitions(thread-pool PRIVATE WIZNOTE_THREAD_POOL_TEST_HOOKS)
wiz_unit(async-api-lifetime async-api-lifetime.cpp)
target_sources(async-api-lifetime PRIVATE "${root}/src/sync/WizAsyncApi.cpp" "${root}/src/sync/WizAsyncApi.h" "${root}/src/share/WizThreads.cpp" "${root}/src/share/WizThreads_p.h")
target_include_directories(async-api-lifetime PRIVATE "${root}/src/sync" "${compat}/fixtures")
target_compile_definitions(async-api-lifetime PRIVATE WIZNOTE_ASYNC_API_TEST_TRANSPORT)
target_link_libraries(async-api-lifetime PRIVATE Qt6::Network)
wiz_unit(document-async document-async.cpp)
wiz_unit(pdf-export-lifecycle pdf-export-lifecycle.cpp)
wiz_unit(pdf-paper-page pdf-paper-page.cpp)
foreach(name document-async pdf-export-lifecycle pdf-paper-page)
  target_link_libraries(${name} PRIVATE Qt6::WebEngineCore Qt6::WebEngineWidgets)
endforeach()
wiz_unit(login-window-recovery login-window-recovery.cpp)

# normal client startup target is imported from the historical compat project.
