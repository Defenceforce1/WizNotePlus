cmake_minimum_required(VERSION 3.24)
project(WizUIIdleRegression LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(WIZ_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Current or frozen own source root")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${WIZ_SOURCE_ROOT}/src/gui/documentviewer/WizEditorToolBar.cpp" "${WIZ_SOURCE_ROOT}/src/gui/categoryviewer/WizCategoryView.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${WIZ_SOURCE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_executable(ui-idle main.cpp)
target_include_directories(ui-idle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(ui-idle PRIVATE cxx_std_17)
target_link_libraries(ui-idle PRIVATE Qt6::Core Qt6::Widgets)
