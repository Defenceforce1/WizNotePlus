cmake_minimum_required(VERSION 3.24)
project(StandaloneNativeHeader LANGUAGES CXX OBJCXX)
find_package(Qt6 6.9 REQUIRED COMPONENTS Widgets)

# Compile the actual narrow implementation, not a separately maintained mock.
# Nothing else from Wiz (identity, editor, database or WebEngine) is linked.
set(header_source "${CMAKE_CURRENT_SOURCE_DIR}/../../src/mac/WizMacHelper.mm")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${header_source}")
file(READ "${header_source}" source)
string(REPLACE "\r\n" "\n" source "${source}")
string(FIND "${source}" "void wizMacConfigureUnifiedHeader(" begin)
string(FIND "${source}" "\nQString WizMacGetOSVersion()" end)
if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "Cannot locate the exact unified-header implementation")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${source}" ${begin} ${length} implementation)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header-under-test.inc" "${implementation}")
file(SHA256 "${header_source}" source_sha)
set(candidate_source "${CMAKE_CURRENT_SOURCE_DIR}/candidate-observer.inc")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${candidate_source}")
file(SHA256 "${candidate_source}" candidate_sha)
add_executable(native-header-probe MACOSX_BUNDLE main.mm)
target_compile_features(native-header-probe PRIVATE cxx_std_17)
target_include_directories(native-header-probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(native-header-probe PRIVATE WIZ_HEADER_SOURCE_SHA="${source_sha}")
target_compile_definitions(native-header-probe PRIVATE WIZ_HEADER_CANDIDATE_SHA="${candidate_sha}")
target_link_libraries(native-header-probe PRIVATE Qt6::Widgets "-framework AppKit")
set_target_properties(native-header-probe PROPERTIES
  MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.defenceforce1.wiznoteplus.tests.native-header"
  MACOSX_BUNDLE_BUNDLE_NAME "Wiz Native Header Probe")
option(WIZ_HEADER_SANITIZE "Instrument the extracted implementation and probe" OFF)
if(WIZ_HEADER_SANITIZE)
  target_compile_options(native-header-probe PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(native-header-probe PRIVATE -fsanitize=address,undefined)
endif()
