cmake_minimum_required(VERSION 3.24)
project(WizQuaZipUpgrade LANGUAGES C CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Core5Compat)
set(WIZ_QUAZIP_SOURCE "" CACHE PATH "Explicit isolated QuaZip source")
set(WIZ_QUAZIP_SUPPORT_OBJECTS "" CACHE PATH "Compatible production support objects")
if(NOT EXISTS "${WIZ_QUAZIP_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Explicit QuaZip source directory required")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static isolated candidate")
set(QUAZIP_INSTALL OFF CACHE BOOL "No install")
set(QUAZIP_ENABLE_TESTS OFF CACHE BOOL "Only focused application wrapper test")
set(QUAZIP_BZIP2 OFF CACHE BOOL "Keep existing zlib-only behavior")
set(QUAZIP_ENABLE_QTEXTCODEC ON CACHE BOOL "Keep existing encoding backend")
set(QUAZIP_FETCH_LIBS OFF CACHE BOOL "No dependency downloads during build")
set(QUAZIP_FORCE_FETCH_LIBS OFF CACHE BOOL "No dependency downloads")
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Offline build")
add_subdirectory("${WIZ_QUAZIP_SOURCE}" dependency)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(quazip-wrapper wrapper.cpp "${root}/src/share/WizZip.cpp"
  "${WIZ_QUAZIP_SUPPORT_OBJECTS}/share/WizMisc.cpp.o"
  "${WIZ_QUAZIP_SUPPORT_OBJECTS}/share/WizQtHelper.cpp.o"
  "${WIZ_QUAZIP_SUPPORT_OBJECTS}/utils/WizMisc.cpp.o")
target_compile_features(quazip-wrapper PRIVATE cxx_std_17)
target_include_directories(quazip-wrapper PRIVATE "${root}/src")
target_link_libraries(quazip-wrapper PRIVATE QuaZip Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Core5Compat)
get_target_property(WIZ_QUAZIP_COMPONENT_VERSION QuaZip VERSION)
target_compile_definitions(quazip-wrapper PRIVATE WIZ_QUAZIP_VERSION="${WIZ_QUAZIP_COMPONENT_VERSION}")
target_link_options(quazip-wrapper PRIVATE -Wl,-dead_strip)
