cmake_minimum_required(VERSION 3.24)
project(StandaloneMessageBox LANGUAGES CXX)
find_package(Qt6 6.9 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
set(actual "${CMAKE_CURRENT_SOURCE_DIR}/../../src/share/WizMessageBox.cpp")
set(header "${CMAKE_CURRENT_SOURCE_DIR}/../../src/share/WizMessageBox.h")
set(candidate "${CMAKE_CURRENT_SOURCE_DIR}/candidate-body.inc")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${actual}" "${header}" "${candidate}")
file(READ "${actual}" source)
string(REPLACE "\r\n" "\n" source "${source}")
string(FIND "${source}" "QMessageBox::StandardButton WizMessageBox::messageBox(" begin)
string(FIND "${source}" "void WizMessageBox::resizeEvent(" end)
if(begin LESS 0 OR end LESS begin OR NOT source MATCHES "WizMessageBox msg\\(parent\\);")
  message(FATAL_ERROR "Actual stack-dialog implementation changed; review this diagnostic first")
endif()
string(SUBSTRING "${source}" 0 ${begin} prefix)
string(SUBSTRING "${source}" ${end} -1 suffix)
file(READ "${candidate}" replacement)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/candidate-message-box.cpp" "${prefix}\n${replacement}\n${suffix}")
file(SHA256 "${actual}" source_sha)
file(SHA256 "${candidate}" candidate_sha)
option(WIZ_MESSAGE_BOX_SANITIZE "Instrument own harness/helper; Qt remains uninstrumented" ON)
foreach(mode baseline candidate)
  if(mode STREQUAL "baseline")
    set(implementation "${actual}")
  else()
    set(implementation "${CMAKE_CURRENT_BINARY_DIR}/candidate-message-box.cpp")
  endif()
  add_executable(message-box-${mode} main.cpp "${implementation}" "${header}")
  target_include_directories(message-box-${mode} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/share")
  target_link_libraries(message-box-${mode} PRIVATE Qt6::Widgets)
  target_compile_features(message-box-${mode} PRIVATE cxx_std_17)
  target_compile_definitions(message-box-${mode} PRIVATE
    WIZ_MESSAGE_BOX_MODE="${mode}" WIZ_MESSAGE_BOX_SOURCE_SHA="${source_sha}"
    WIZ_MESSAGE_BOX_CANDIDATE_SHA="${candidate_sha}")
  if(WIZ_MESSAGE_BOX_SANITIZE)
    target_compile_options(message-box-${mode} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(message-box-${mode} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
